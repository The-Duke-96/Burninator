# ==============================================================================
# BURNINATOR v3 — Automated Matrix Export Script
# ==============================================================================
# Drives Fusion 360's parametric model through a matrix of hotend lengths and
# extruder screw-hole positions, applies temporary boolean cuts for hole
# patterns, and exports each variant as a high-resolution STL.
#
# BOOLEAN-CUT STRATEGY:
#   Cowls need different hole patterns per extruder/fitting. Instead of
#   duplicating geometry, this script temporarily applies a CombineFeature
#   (Cut) using invisible tool bodies, exports the result, then deletes the
#   feature — leaving the base body intact for the next variant.
#   isKeepToolBodies = True ensures tool bodies are never consumed.
#
# ADDING NEW VARIANTS:
#   Hotend length  → HOTEND_CONFIGS  |  Extruder type → SCREW_CONFIGS
#   Static body    → STATIC_BODIES   |  Parametric part → HOTEND_DEPENDENT_BODIES
#   No changes to run() required.
#
# NOTE: 'hotend_lenght' is an intentional typo — it matches the parameter name
#       in the Fusion 360 design file. Do not correct it here.
# ==============================================================================

# ==============================================================================
# HELPER FUNCTIONS
# ==============================================================================

def get_body_by_path(root_comp, path_str):
    """
    Return a BRep body by its slash-separated occurrence path.
    Returns None (with a print warning) if any segment is not found,
    so a missing body skips its export without aborting the whole run.
    """
    try:
        parts = path_str.split('/')
        *occ_parts, body_name = parts
        current_occ = None
        for part in occ_parts:
            occurrences = root_comp.occurrences if current_occ is None else current_occ.childOccurrences
            current_occ = occurrences.itemByName(part)
            if not current_occ:
                print(f"WARNING: Occurrence not found: '{part}' in path '{path_str}'")
                return None
        body = current_occ.component.bRepBodies.itemByName(body_name)
        if not body:
            print(f"WARNING: Body not found: '{body_name}' in path '{path_str}'")
        return body
    except Exception as e:
        print(f"ERROR: get_body_by_path('{path_str}'): {e}")
        return None


def cut_body_with_tools(root_comp, target_body, tool_bodies):
    """
    Boolean-cut target_body with tool_bodies and return the CombineFeature.
    isKeepToolBodies = True prevents the tool solids from being consumed,
    so they remain available for subsequent cuts in the export loop.
    The caller must call cut_feat.deleteMe() after exporting to restore the body.
    """
    valid_tools = [t for t in tool_bodies if t is not None]
    if not target_body or not valid_tools:
        print("WARNING: cut_body_with_tools — missing target or tools, skipping.")
        return None
    tool_collection = adsk.core.ObjectCollection.create()
    for tool in valid_tools:
        tool_collection.add(tool)
    combine_input = root_comp.features.combineFeatures.createInput(target_body, tool_collection)
    combine_input.operation = adsk.fusion.FeatureOperations.CutFeatureOperation
    combine_input.isKeepToolBodies = True
    return root_comp.features.combineFeatures.add(combine_input)


def export_stl(export_mgr, body, folder_key, filename):
    """Export a single body to STL at high mesh refinement."""
    out_path = os.path.join(FOLDERS[folder_key], filename)
    stl_options = export_mgr.createSTLExportOptions(body, out_path)
    stl_options.meshRefinement = adsk.fusion.MeshRefinementSettings.MeshRefinementHigh
    export_mgr.execute(stl_options)
    print(f"Exported: {out_path}")


def set_param(param, expression):
    """
    Set a user parameter and yield control twice so Fusion's async model
    regeneration has time to complete before the next operation reads geometry.
    """
    param.expression = expression
    adsk.doEvents()
    adsk.doEvents()


def cut_export_restore(root_comp, export_mgr, body, tools, folder_key, filename):
    """
    Apply a boolean cut, export the result, then delete the cut feature to
    restore the body for the next variant. If the cut couldn't be created,
    the body is exported uncut and a warning is printed.
    """
    cut_feat = cut_body_with_tools(root_comp, body, tools)
    export_stl(export_mgr, body, folder_key, filename)
    if cut_feat:
        cut_feat.deleteMe()
    else:
        print(f"WARNING: No cut applied for '{filename}' — exported without hole pattern.")


# ==============================================================================
# ENTRY POINT
# ==============================================================================

def run(context):
    """
    Main entry point called by Fusion 360.

    Export order:
        1. CAD master files (.step / .f3d)
        2. Static bodies (no parameter dependency)
        3. SF-cowl + LGX adapter  — hotend fixed at 54 mm, extruder matrix (×3)
        4. UHF-cowl + dependent parts — full hotend (×4) × extruder (×3) matrix
    """
    ui = None
    try:
        app    = adsk.core.Application.get()
        ui     = app.userInterface
        design = app.activeProduct

        if not design:
            ui.messageBox('No active design found.')
            return

        root_comp  = design.rootComponent
        export_mgr = design.exportManager

        # Fetch the three driving parameters and abort early if any are missing
        param_hotend  = design.userParameters.itemByName('hotend_lenght')          # "lenght" typo matches the design file
        param_screw_r = design.userParameters.itemByName('extruder_screwholes_right')
        param_screw_l = design.userParameters.itemByName('extruder_screwholes_left')

        missing = [name for name, p in [
            ('hotend_lenght',             param_hotend),
            ('extruder_screwholes_right', param_screw_r),
            ('extruder_screwholes_left',  param_screw_l),
        ] if p is None]

        if missing:
            ui.messageBox(f"Aborted: User parameter(s) not found:\n  {', '.join(missing)}")
            return

        # Create the complete folder tree before writing any files
        for folder_path in FOLDERS.values():
            os.makedirs(folder_path, exist_ok=True)

        # -----------------------------------------------------------------------
        # 1. Master CAD files — exported once from the root component
        # -----------------------------------------------------------------------
        export_mgr.execute(export_mgr.createSTEPExportOptions(
            os.path.join(FOLDERS["cad"], "Burninator_v3.step"), root_comp))

        export_mgr.execute(export_mgr.createFusionArchiveExportOptions(
            os.path.join(FOLDERS["cad"], "Burninator_v3.f3d")))

        # -----------------------------------------------------------------------
        # 2. Static body exports — no parameter changes needed
        # -----------------------------------------------------------------------
        for path, (folder_key, filename) in STATIC_BODIES.items():
            body = get_body_by_path(root_comp, path)
            if body:
                export_stl(export_mgr, body, folder_key, filename)

        # -----------------------------------------------------------------------
        # Boolean-cut tool bodies — invisible solids used only as cutting tools.
        # Positions are parameter-driven, so they move with screw-hole updates.
        # -----------------------------------------------------------------------
        body_ecas    = get_body_by_path(root_comp, "Extruder:1/ecas:1/ecas")
        body_screw_r = get_body_by_path(root_comp, "Extruder:1/screwholes:1/right_screwhole")
        body_screw_l = get_body_by_path(root_comp, "Extruder:1/screwholes:1/left_screwhole")

        tools_ecas       = [body_ecas]
        tools_screwholes = [body_screw_r, body_screw_l]

        # -----------------------------------------------------------------------
        # 3. SF-cowl and LGX adapter
        #    Hotend length is fixed at 54 mm; iterates over extruder screw configs.
        #    Output: SF-cowl_ECAS + SF-cowl_<extruder> (×3) + lgx_adapter_<extruder> (×3)
        # -----------------------------------------------------------------------
        set_param(param_hotend, "54.0mm")

        body_sf = get_body_by_path(root_comp, "Cowls:1/SF:1/SF_cowl")

        # ECAS variant — ECAS pocket only, no screw holes
        if body_sf:
            cut_export_restore(root_comp, export_mgr, body_sf, tools_ecas,
                               "cowls_ecas", "SF-cowl_ECAS.stl")

        # Extruder screw-hole variants — one per extruder, no ECAS pocket.
        # body_lgx is re-fetched each iteration because deleteMe() can invalidate
        # body references in Fusion's history after the previous cut was rolled back.
        for s_cfg in SCREW_CONFIGS:
            set_param(param_screw_r, s_cfg["r"])
            set_param(param_screw_l, s_cfg["l"])
            if body_sf:
                cut_export_restore(root_comp, export_mgr, body_sf, tools_screwholes,
                                   s_cfg["folder"], f"SF-cowl_{s_cfg['name']}.stl")
            body_lgx = get_body_by_path(root_comp, "Extruder:1/LGX_adapters:1/LGX_adapter")
            if body_lgx:
                cut_export_restore(root_comp, export_mgr, body_lgx, tools_screwholes,
                                   "cowls_lgx", f"lgx_adapter_{s_cfg['name']}.stl")

        # -----------------------------------------------------------------------
        # 4. UHF-cowl, backplates, docks, and rail carriages
        #    Full matrix: hotend length (×4) × extruder screw config (×3)
        #    Dependent bodies (docks, backplates, carriages) are exported once per
        #    hotend length — their geometry updates automatically via parameters.
        # -----------------------------------------------------------------------
        body_uhf = get_body_by_path(root_comp, "Cowls:1/UHF:1/UHF_cowl")

        for h_cfg in HOTEND_CONFIGS:
            set_param(param_hotend, h_cfg["val"])

            # UHF-cowl: ECAS variant for this hotend length
            if body_uhf:
                cut_export_restore(root_comp, export_mgr, body_uhf, tools_ecas,
                                   "cowls_ecas", f"{h_cfg['name']}-cowl_ECAS.stl")

            # UHF-cowl: extruder screw-hole variants for this hotend length
            if body_uhf:
                for s_cfg in SCREW_CONFIGS:
                    set_param(param_screw_r, s_cfg["r"])
                    set_param(param_screw_l, s_cfg["l"])
                    cut_export_restore(root_comp, export_mgr, body_uhf, tools_screwholes,
                                       s_cfg["folder"], f"{h_cfg['name']}-cowl_{s_cfg['name']}.stl")

            # Hotend-dependent bodies: docks, backplates, carriages.
            # The 54 mm length is shared between SF and UHF cowls, so it gets
            # the combined suffix "SF+UHF" to reflect both use cases.
            version_suffix = "SF+UHF" if h_cfg["name"] == "UHF" else h_cfg["name"]

            for path, (folder_key, prefix) in HOTEND_DEPENDENT_BODIES.items():
                body = get_body_by_path(root_comp, path)
                if body:
                    export_stl(export_mgr, body, folder_key, f"{prefix}_{version_suffix}.stl")

    except Exception:
        if ui:
            ui.messageBox(f"Script execution failed:\n{traceback.format_exc()}")
        return

    ui.messageBox('Burninator_v3: Automated Matrix Export Completed Successfully!')
